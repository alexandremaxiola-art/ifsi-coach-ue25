async function validate(method,body){

const topic=(body?.topic||'').trim(); if(!topic) return json({error:'topic_required'},400);
list=list.includes(topic)?list.filter(x=>x!==topic):[...list,topic];
const ts=now();
await rest('ue25_progress?student_id=eq.'+encodeURIComponent(part.id),{method:'PATCH',headers:{Prefer:'return=minimal'},body:JSON.stringify({validated_modules:list,last_seen_at:ts,updated_at:ts})});

async function loadValidationMap(force=false){
const part=getLocal(); if(!part?.id) return;
if(!force && validationLoadedFor===part.id) return;
const rows=await rest('ue25_progress?select=module_validations,validated_modules&student_id=eq.'+encodeURIComponent(part.id)+'&limit=1');
const p=rows?.[0]||{};
validationMap=(p.module_validations&&typeof p.module_validations==='object')?structuredClone(p.module_validations):{};
legacyValidated=Array.isArray(p.validated_modules)?[...p.validated_modules]:[];
for(const topic of legacyValidated){
validationMap[topic]=validationMap[topic]||{};
if(validationMap[topic].revise!==true) validationMap[topic].revise=true;
}
validationLoadedFor=part.id;
refreshValidationUI();
}
function validationDone(topic,type){return !!validationMap?.[topic]?.[type]}
async function togglePartValidation(topic,type){
const part=getLocal(); if(!part?.id||!topic) return;
await loadValidationMap();
const next=structuredClone(validationMap||{});
next[topic]=next[topic]||{};
next[topic][type]=!next[topic][type];
const ts=now();
const allDone=['test','revise','case'].every(t=>!!next?.[topic]?.[t]);
const legacyNext=allDone?[...new Set([...legacyValidated,topic])]:legacyValidated.filter(x=>x!==topic);
await rest('ue25_progress?student_id=eq.'+encodeURIComponent(part.id),{method:'PATCH',headers:{Prefer:'return=minimal'},body:JSON.stringify({module_validations:next,validated_modules:legacyNext,last_seen_at:ts,updated_at:ts})});
validationMap=next;legacyValidated=legacyNext;
refreshValidationUI();
}
